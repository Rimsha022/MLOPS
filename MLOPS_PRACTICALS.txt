numeric_columns = df.select_dtypes(include=np.number).columns

for col in numeric_columns:
    df[col] = df[col].fillna(df[col].median())

df.isnull().sum()

df["High_Temperature"] = np.where(
    df["Temperature_C"] > 60, 1, 0
)

df["Low_Battery"] = np.where(
    df["Battery_Voltage"] < 45, 1, 0
)

df["Power_Traffic_Ratio"] = (
    df["Power_Consumption_W"] / (df["Traffic_Load"] + 0.01)
)

df.head()

df["Tower_Age_Category"] = pd.cut(
    df["Tower_Age_Years"],
    bins=[0, 3, 7, float("inf")],
    labels=["New", "Medium", "Old"]
)

df[["Tower_Age_Years", "Tower_Age_Category"]].head(10)

df = pd.get_dummies(
    df,
    columns=["Tower_Age_Category"],

import yaml
import joblib

from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score

# Step 1: Load YAML configuration
with open("config.yml", "r") as file:
    config = yaml.safe_load(file)

print("Project:", config["project_name"])

# Step 2: Load the dataset
data = load_breast_cancer()
X = data.data
y = data.target



# Step 3: Split data using YAML settings
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=config["training"]["test_size"],
    random_state=config["model"]["random_state"]
)

# Step 4: Create model using YAML parameters
model = RandomForestClassifier(
    n_estimators=config["model"]["n_estimators"],
    max_depth=config["model"]["max_depth"],
    random_state=config["model"]["random_state"]
)

# Step 5: Train the model
model.fit(X_train, y_train)

# Step 6: Evaluate the model
predictions = model.predict(X_test)
accuracy = accuracy_score(y_test, predictions)

print("Algorithm:", config["model"]["algorithm"])
print("Number of trees:", config["model"]["n_estimators"])
print("Test data proportion:", config["training"]["test_size"])
print("Model Accuracy:", round(accuracy * 100, 2), "%")

# Step 7: Save the trained model
joblib.dump(model, "model.pkl")
print("Model saved successfully as model.pkl")
